cmake_minimum_required(VERSION 3.20)
project(ShortcutPresetTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_executable(shortcut-preset-tests main.cc)
target_link_libraries(shortcut-preset-tests PRIVATE Qt6::Widgets)
enable_testing()
add_test(NAME shortcut-preset-tests COMMAND shortcut-preset-tests)
set_tests_properties(shortcut-preset-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
