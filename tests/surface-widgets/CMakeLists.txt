cmake_minimum_required(VERSION 3.20)
project(SurfaceWidgetTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_executable(surface-widget-tests main.cc)
target_link_libraries(surface-widget-tests PRIVATE Qt6::Widgets)
enable_testing()
add_test(NAME surface-widget-tests COMMAND surface-widget-tests)
set_tests_properties(surface-widget-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
